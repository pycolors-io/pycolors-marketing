import type { MetadataRoute } from "next";

import { pwaConfig } from "@/lib/pwa/pwa-config";

    id: "/",
    name: pwaConfig.name,
    short_name: pwaConfig.shortName,
    description: pwaConfig.description,

    scope: "/",

    background_color: pwaConfig.backgroundColor,
    categories: ["business", "productivity", "developer"],

      {
        src: "/pwa/icon-192.png",
        sizes: "192x192",

      },

</Step>

  <Step>
    ### Inscrire le travailleur de service [#register-the-service-worker]

```tsx title="components/pwa/pwa-register.tsx"
"use client";

import { useEffect } from "react";

export function PwaRegister() {
  useEffect(() => {
    if (!("serviceWorker" in navigator)) {
      return;
    }

    const registerServiceWorker = async () => {
      try {
        await navigator.serviceWorker.register("/sw.js");
      } catch {
        // PWA enhancements must never break the core app.
      }
    };

    void registerServiceWorker();
  }, []);

  return null;
}
```

</Step>

## Questions courantes [#common-questions]

<Accordions type="single">
  <Accordion title="Dois-je utiliser next-pwa immédiatement?">
    Pas nécessairement. Starter Pro utilise d'abord une base PWA explicite car il est plus facile à auditer et plus sûr pour les données SaaS protégées.
  </Accordion>

  <Accordion title="Les pages de facturation devraient-elles fonctionner hors ligne?">
    Non. La facturation doit rester en ligne d'abord parce que l'état de Stripe, l'accès à l'abonnement et les données de facture doivent rester vérifiables par le serveur.
  </Accordion>
</Accordions>
